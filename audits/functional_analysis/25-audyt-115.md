```python
from fractions import Fraction as F
from itertools import product
from math import factorial,comb
P=[{(0,0):F(1,3),(1,0):F(-2),(0,2):F(3,4)}, {(2,1):F(2,3),(1,1):F(1),(0,0):F(-1)}, {(0,3):F(1,2),(3,0):F(-1,4),(1,2):F(2)}]
X=[(F(0),F(0)),(F(1,2),F(-1,3)),(F(-2),F(3,4)),(F(2,3),F(1))]
A=[(i,j) for i in range(4) for j in range(4-i)]
def mul(p,q):
    r={}
    for (i,j),a in p.items():
        for (k,l),b in q.items():
            key=(i+k,j+l);r[key]=r.get(key,F(0))+a*b
    return r
def jet(p,x,a):
    k,l=a;u,v=x
    return sum(c*F(factorial(i),factorial(i-k))*F(factorial(j),factorial(j-l))*u**(i-k)*v**(j-l) for (i,j),c in p.items() if i>=k and j>=l)
c=[0,0]
for p,q,x,a in product(P,P,X,A):
    k,l=a;m=k+l; d=jet(mul(p,q),x,a)
    rhs=sum(comb(k,i)*comb(l,j)*jet(p,x,(i,j))*jet(q,x,(k-i,l-j)) for i in range(k+1) for j in range(l+1))
    assert d==rhs;c[0]+=1
    z=[b for b in A if sum(b)<=m]
    assert abs(d)<=2**m*max(abs(jet(p,x,b)) for b in z)*max(abs(jet(q,x,b)) for b in z);c[1]+=1
print(c,sum(c))
```
